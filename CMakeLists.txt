cmake_minimum_required(VERSION 3.24)
project(Muse VERSION 0.5.1 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Muse currently targets Windows 10/11 x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Single source of truth for the version shown in the UI, the HTTP user agent and
# the executable's version resource.
add_compile_definitions(MUSE_VERSION="${PROJECT_VERSION}")
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_MakeAvailable(json)
add_library(muse_engine src/config.cpp src/platform.cpp src/engine.cpp src/metadata.cpp)
target_include_directories(muse_engine PUBLIC src)
target_link_libraries(muse_engine PUBLIC nlohmann_json::nlohmann_json winhttp wininet crypt32 bcrypt advapi32 shell32 ole32 comdlg32 dwmapi uuid)
target_compile_definitions(muse_engine PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2)
qt_standard_project_setup()
configure_file(src/muse.rc.in ${CMAKE_CURRENT_BINARY_DIR}/muse.rc @ONLY)
set(MUSE_UI_SOURCES
  src/qt_bridge.cpp src/qt_bridge.hpp
  src/app_icons.cpp src/app_icons.hpp
  src/app_catalog.cpp src/app_catalog.hpp
  src/app_library.cpp src/app_library.hpp
  src/muse.qrc)
qt_add_executable(Muse WIN32 src/qt_main.cpp ${MUSE_UI_SOURCES} ${CMAKE_CURRENT_BINARY_DIR}/muse.rc)
set_target_properties(Muse PROPERTIES AUTOMOC ON AUTORCC ON)
target_link_libraries(Muse PRIVATE muse_engine Qt6::Quick Qt6::QuickControls2 comctl32 dwmapi version)
if(MSVC)
  foreach(t muse_engine Muse)
    target_compile_options(${t} PRIVATE /utf-8 /W4 /EHsc)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(muse_tests tests/config_tests.cpp)
  target_link_libraries(muse_tests PRIVATE muse_engine)
  add_executable(muse_ui_tests tests/ui_tests.cpp ${MUSE_UI_SOURCES})
  target_link_libraries(muse_ui_tests PRIVATE muse_engine Qt6::Quick comctl32 version)
  add_executable(muse_qml_tests tests/qml_tests.cpp ${MUSE_UI_SOURCES})
  set_target_properties(muse_qml_tests PROPERTIES AUTOMOC ON AUTORCC ON)
  target_link_libraries(muse_qml_tests PRIVATE muse_engine Qt6::Quick Qt6::QuickControls2 comctl32 version)
  foreach(t muse_tests muse_ui_tests muse_qml_tests)
    if(MSVC)
      target_compile_options(${t} PRIVATE /utf-8 /W4 /EHsc)
    endif()
  endforeach()

  add_test(NAME config_and_storage COMMAND muse_tests)
  add_test(NAME selection_and_icons COMMAND muse_ui_tests)
  set_tests_properties(selection_and_icons PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
  add_test(NAME qml_clicks_and_layout COMMAND muse_qml_tests)
  set_tests_properties(qml_clicks_and_layout PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/core/mihomo.exe")
    add_test(NAME core_integration COMMAND muse_tests --core "${CMAKE_CURRENT_SOURCE_DIR}/core/mihomo.exe")
  endif()
endif()
install(TARGETS Muse RUNTIME DESTINATION .)
